#!/usr/bin/env python3 # -*- coding: utf-8 -*- from __future__ import annotations import secrets import subprocess import threading import time from pathlib import Path from fastapi import ( FastAPI, HTTPException, Request, Response, ) from fastapi.responses import ( HTMLResponse, StreamingResponse, FileResponse, ) from pydantic import BaseModel from core.robot_core import RobotCore from database.database import UserDatabase from camera.camera import CameraManager from voice.audio import AudioManager # ============================================================= # 测试音乐 # ============================================================= TEST_AUDIO_DIR = Path( "/home/henggui/robot/voice/test_audio" ) TEST_MUSIC_MP3 = ( TEST_AUDIO_DIR / "wake.mp3" ) TEST_MUSIC_WAV = ( TEST_AUDIO_DIR / "wake.wav" ) # 不使用 card 3。 # 使用声卡名称，避免重启后声卡编号改变。 TEST_AUDIO_DEVICE = ( "plughw:CARD=seeed2micvoicec,DEV=0" ) test_music_lock = threading.RLock() test_music_process = None def test_music_is_playing(): global test_music_process with test_music_lock: if test_music_process is None: return False if test_music_process.poll() is None: return True test_music_process = None return False def stop_test_music_process(): global test_music_process with test_music_lock: process = test_music_process if process is None: return True if process.poll() is None: try: process.terminate() process.wait( timeout=1.5 ) except subprocess.TimeoutExpired: try: process.kill() process.wait( timeout=1.0 ) except Exception: pass except Exception: pass test_music_process = None return True def ensure_test_music_wav(): """ 如果 wake.wav 已存在，直接使用。 如果只有 wake.mp3，则第一次播放时自动转换： MP3 -> 48 kHz / Stereo WAV """ TEST_AUDIO_DIR.mkdir( parents=True, exist_ok=True, ) if TEST_MUSIC_WAV.exists(): return TEST_MUSIC_WAV if not TEST_MUSIC_MP3.exists(): raise FileNotFoundError( "测试音乐不存在：" + str(TEST_MUSIC_MP3) ) print( "[AUDIO] Converting:", TEST_MUSIC_MP3, "->", TEST_MUSIC_WAV, ) try: result = subprocess.run( [ "ffmpeg", "-y", "-loglevel", "error", "-i", str(TEST_MUSIC_MP3), "-ar", "48000", "-ac", "2", str(TEST_MUSIC_WAV), ], capture_output=True, text=True, timeout=120, ) except FileNotFoundError: raise RuntimeError( "系统未安装 ffmpeg，" "无法将 wake.mp3 转换为 WAV" ) except subprocess.TimeoutExpired: raise RuntimeError( "wake.mp3 转换超时" ) if result.returncode != 0: raise RuntimeError( "音频转换失败：" + ( result.stderr.strip() or "ffmpeg error" ) ) if not TEST_MUSIC_WAV.exists(): raise RuntimeError( "转换完成后未找到 wake.wav" ) return TEST_MUSIC_WAV def play_test_music_process(): global test_music_process # --------------------------------------------------------- # 确保测试音频存在 # --------------------------------------------------------- path = ensure_test_music_wav() # --------------------------------------------------------- # 先停止已有测试音乐 # --------------------------------------------------------- stop_test_music_process() # --------------------------------------------------------- # 再停止 AudioManager 当前播放 # # 避免两个 aplay 同时占用同一块 ALSA 声卡 # --------------------------------------------------------- try: audio.stop_playback() except Exception: pass # --------------------------------------------------------- # 直接指定 Seeed 声卡播放 # --------------------------------------------------------- with test_music_lock: try: test_music_process = subprocess.Popen( [ "aplay", "-D", TEST_AUDIO_DEVICE, str(path), ], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, ) except FileNotFoundError: test_music_process = None raise RuntimeError( "系统没有找到 aplay" ) except Exception as exc: test_music_process = None raise RuntimeError( "启动测试音乐失败：" + str(exc) ) # --------------------------------------------------------- # 给 aplay 一点时间。 # 如果立即退出，通常说明声卡打开失败。 # --------------------------------------------------------- time.sleep(0.15) with test_music_lock: if ( test_music_process is None or test_music_process.poll() is not None ): test_music_process = None raise RuntimeError( "aplay 启动后立即退出。" "请检查 seeed2micvoicec " "声卡、喇叭或 ALSA 配置。" ) return { "ok": True, "playing": True, "filename": path.name, "device": TEST_AUDIO_DEVICE, } # ============================================================= # Session # ============================================================= sessions = {} session_lock = threading.RLock() # ============================================================= # API 数据模型 # ============================================================= class LoginBody(BaseModel): username: str password: str class SpeedBody(BaseModel): speed: int = 80 class HeadBody(BaseModel): angle: int class ArmBody(BaseModel): joint: int angle: int class GripperBody(BaseModel): angle: int class AutoBody(BaseModel): enabled: bool class RawBody(BaseModel): command: str class RecordBody(BaseModel): seconds: int = 5 class VolumeBody(BaseModel): volume: int = 50 #
